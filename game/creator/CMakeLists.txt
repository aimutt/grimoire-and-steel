# Shared shop item-art catalog, baked into the Module Creator binary (same images the
# Game Engine renders at play time), referenced by filename id.
gns_generate_item_rc(
    "${CMAKE_CURRENT_BINARY_DIR}/embedded_items.rc"
    "${CMAKE_CURRENT_BINARY_DIR}/embedded_items.h")

# Bake the shared Grimoire & Steel splash (the same PNG the Game Engine shows on startup)
# into the binary as a GNSSPLASH RCDATA resource, loaded via gns::ui::loadEmbeddedTexture.
set(_splash "${CMAKE_SOURCE_DIR}/engine/assets/gns-splash.png")
string(REPLACE "/" "\\\\" _splashrc "${_splash}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/embedded_splash.rc" "GNSSPLASH RCDATA \"${_splashrc}\"\n")

# Bake the SAME character portraits the Game Engine uses (tracked under engine/assets/portraits)
# so the shared character builder can offer a portrait picker here too. Generates a .rc +
# a kEmbeddedPortraits header identical in shape to the engine's.
file(GLOB CREATOR_PORTRAITS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/engine/assets/portraits/*.png")
list(SORT CREATOR_PORTRAITS)
set(_prc "// generated by CMake -- do not edit\n")
set(_phdr "// generated by CMake -- do not edit\n#pragma once\n#include <string>\n#include <vector>\n")
string(APPEND _phdr "struct EmbeddedPortrait { const char* file; const char* res; };\n")
string(APPEND _phdr "inline const std::vector<EmbeddedPortrait> kEmbeddedPortraits = {\n")
foreach(png ${CREATOR_PORTRAITS})
    get_filename_component(_fname "${png}" NAME)
    get_filename_component(_stem "${png}" NAME_WE)
    string(TOUPPER "${_stem}" _id)
    string(REPLACE "/" "\\\\" _rcpath "${png}")
    string(APPEND _prc "${_id} RCDATA \"${_rcpath}\"\n")
    string(APPEND _phdr "    { \"${_fname}\", \"${_id}\" },\n")
endforeach()
string(APPEND _phdr "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/embedded_portraits.rc" "${_prc}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/embedded_portraits.h" "${_phdr}")

add_executable(ModuleCreator src/main.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/embedded_items.rc"
    "${CMAKE_CURRENT_BINARY_DIR}/embedded_splash.rc"
    "${CMAKE_CURRENT_BINARY_DIR}/embedded_portraits.rc")
target_include_directories(ModuleCreator PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ModuleCreator PRIVATE
    gns_core gns_ui imgui SDL2::SDL2 SDL2::SDL2main SDL2_image::SDL2_image)
if (WIN32)
    target_link_libraries(ModuleCreator PRIVATE comdlg32)   # native file dialogs
endif()
gns_copy_runtime(ModuleCreator)
